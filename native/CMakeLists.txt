cmake_minimum_required(VERSION 3.31)
project(WindropNative LANGUAGES C CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The native backend currently targets MSVC x64 on Windows")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
foreach(path DEVOURER_SOURCE OWL_SOURCE LIBUSB_SOURCE WINTUN_SOURCE)
    file(TO_CMAKE_PATH "${${path}}" normalized)
    set(${path} "${normalized}")
endforeach()

# Chip initialization stays in the separately licensed GPL2 radio executable.
foreach(chip JAGUAR1 8814 JAGUAR2_8822B JAGUAR2_8821C JAGUAR3_8822C JAGUAR3_8822E 8733B KESTREL_8852B MT7612U PCIE)
    set(DEVOURER_${chip} OFF CACHE BOOL "" FORCE)
endforeach()
set(DEVOURER_KESTREL_8852C ON CACHE BOOL "" FORCE)
set(WINDROP_RADIO_ONLY ON)
add_subdirectory("${DEVOURER_SOURCE}" devourer EXCLUDE_FROM_ALL SYSTEM)
add_executable(windrop-hardware hardware/main.cpp)
target_link_libraries(windrop-hardware PRIVATE devourer setupapi)
target_compile_options(windrop-hardware PRIVATE /W4 /WX)

# OWL's protocol core has no dependency on Linux pcap/libev/netlink adapters.
add_library(radiotap STATIC "${OWL_SOURCE}/radiotap/radiotap.c")
target_include_directories(radiotap PUBLIC "${OWL_SOURCE}/radiotap" awdl/compat)
target_compile_options(radiotap PRIVATE /FI"${CMAKE_CURRENT_SOURCE_DIR}/awdl/compat/owl-platform.h")
add_subdirectory("${OWL_SOURCE}/src" owl)
target_include_directories(awdl PUBLIC "${OWL_SOURCE}/src" awdl/compat)
target_compile_options(awdl PRIVATE /FI"${CMAKE_CURRENT_SOURCE_DIR}/awdl/compat/owl-platform.h")
add_executable(windrop-awdl awdl/main.cpp awdl/compat.c)
target_include_directories(windrop-awdl PRIVATE "${WINTUN_SOURCE}/include")
target_link_libraries(windrop-awdl PRIVATE awdl iphlpapi ws2_32 winmm)
target_compile_options(windrop-awdl PRIVATE /W4 /WX)
